export default function Watchlist() {
  const watchlist = useWatchlistStore((state) => state.watchlist);
  const sort = useWatchlistStore((s) => s.sortBy);
  const minRating = useWatchlistStore((s) => s.minRating);
  const setSortBy = useWatchlistStore((s) => s.setSortBy);
  const setMinRating = useWatchlistStore((s) => s.setMinRating);

  const sorted = [...watchlist]
    .filter((m) => m.vote_average >= minRating)
    .sort((a, b) => {
      if (sort === "rating") return b.vote_average - a.vote_average;
      if (sort === "title") return a.title.localeCompare(b.title);
      return new Date(b.addedAt).getTime() - new Date(a.addedAt).getTime();
    });
  if (watchlist.length === 0) {
    return (
      <div className="text-2xl text-center">
        <Link to="/">Home</Link>
        <p>Browse to add some in watchlist</p>
      </div>
    );
  }
  return (
    <>
      <div className="flex flex-wrap mb-8">
        <h1 className="text-3xl font-bold mr-auto">My Watchlist</h1>
        <div className="btn-group flex flex-wrap gap-2.5">
          <label className="flex gap-1 items-center justify-center rounded-lg border border-gray-700 bg-gray-800 px-3 text-sm text-gray-100 placeholder-gray-500 focus:border-indigo-500 focus:outline-none">
            <button
              onClick={() => setSortBy("date")}
              className="py-1.5 pr-2 mr-2 border-r border-gray-700 focus:outline-none"
            >
              {sort === "date" ? <FilterSVG /> : <ClearFilterSVG />}
            </button>
            Sort By:
            <select
              className="bg-gray-800 focus:outline-none"
              value={sort}
              onChange={(e) => setSortBy(e.target.value as SortBy)}
            >
              <option value="date">Date</option>
              <option value="rating">Rating</option>
              <option value="title">Title</option>
            </select>
          </label>
          <label className="flex gap-1 items-center justify-center rounded-lg border border-gray-700 bg-gray-800 px-3 text-sm text-gray-100 placeholder-gray-500 focus:border-indigo-500 focus:outline-none">
            <button
              onClick={() => setMinRating(6)}
              className="py-1.5 pr-2 mr-2 border-r border-gray-700 focus:outline-none"
            >
              <StarRatingSVG />
            </button>
            Min rating:{minRating}+
            <input
              type="range"
              min="4"
              max="8"
              value={minRating}
              onChange={(e) => setMinRating(Number(e.target.value))}
            />
          </label>
        </div>
      </div>
      <div className="grid grid-cols-2 gap-4 sm:grid-cols-3 md:grid-cols-4 lg:grid-cols-5">
        {sorted.map((movie) => (
          <WatchlistCard key={movie.id} movie={movie} />
        ))}
      </div>
    </>
  );
}

useWatchlistStore.ts
import { create } from "zustand";
import { type SortBy, type WatchlistMovie } from "../types/tmdb";
import { persist } from "zustand/middleware";

interface WatchlistStore {
  watchlist: WatchlistMovie[];
  addToWatchlist: (movie: WatchlistMovie) => void;
  removeFromWatchlist: (id: WatchlistMovie["id"]) => void;
  clearWatchlist: () => void;
  sortBy: SortBy;
  minRating: number;
  setSortBy: (sort: SortBy) => void;
  setMinRating: (rating: number) => void;
}

const useWatchlistStore = create<WatchlistStore>()(
  persist(
    (set) => ({
      watchlist: [],
      sortBy: "date",
      minRating: 6,

      addToWatchlist: (movie) =>
        set((state) => ({ watchlist: [...state.watchlist, movie] })),

      removeFromWatchlist: (id) =>
        set((state) => ({
          watchlist: state.watchlist.filter((m) => m.id !== id),
        })),

      clearWatchlist: () => set({ watchlist: [] }),

      setSortBy: (sort) => set({ sortBy: sort }),

      setMinRating: (rate) => set({ minRating: rate }),
    }),
    {
      name: "cinetrack-watchlist",
      partialize: (state) => ({ watchlist: state.watchlist }),
    },
  ),
);

export default useWatchlistStore;
